cmake_minimum_required(VERSION 3.20)
project(QwCore LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(ZLIB REQUIRED)
find_package(Threads REQUIRED)
set(UPSTREAM ${CMAKE_CURRENT_SOURCE_DIR}/../../app/src/main/cpp/jni)
file(GLOB ZSTD_SOURCES ${UPSTREAM}/zstd/lib/common/*.c ${UPSTREAM}/zstd/lib/compress/*.c ${UPSTREAM}/zstd/lib/decompress/*.c)
add_library(qwzstd STATIC ${ZSTD_SOURCES})
target_include_directories(qwzstd PRIVATE ${UPSTREAM}/zstd/lib)
target_compile_definitions(qwzstd PRIVATE ZSTD_DISABLE_ASM=1)
file(GLOB ASTC_SOURCES ${UPSTREAM}/astcenc/Source/astcenc_*.cpp)
add_library(qwastc STATIC ${ASTC_SOURCES})
target_include_directories(qwastc PUBLIC ${UPSTREAM}/astcenc/Source)
target_compile_definitions(qwastc PUBLIC ASTCENC_ISA_NONE=1 ASTCENC_VECALIGN=16)
set(CORE_SOURCES sc_core.cpp sc_ops.cpp sc_ops_patch.cpp tex_ops.cpp png2sc_op.cpp sc2png_exports.cpp inject_impl.cpp mc_copy_impl.cpp sc_sprites_stub.cpp sc_editor_jni.cpp sc2_reader.cpp sc2_replace.cpp gif_sc.cpp build_sc_impl.cpp sctx_parser.cpp astc_decoder.cpp sctool_jni.cpp)
list(TRANSFORM CORE_SOURCES PREPEND "${UPSTREAM}/")
add_library(qwcore STATIC ${CORE_SOURCES} ${UPSTREAM}/stb/stb_impl.c QwCore.cpp)
target_include_directories(qwcore PRIVATE compat ${UPSTREAM} ${UPSTREAM}/zstd/lib ${UPSTREAM}/stb)
target_compile_options(qwcore PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:-include${CMAKE_CURRENT_SOURCE_DIR}/PortablePrefix.h>")
target_link_libraries(qwcore PUBLIC qwzstd qwastc ZLIB::ZLIB Threads::Threads)
option(QW_BUILD_TESTS "Build portable core tests" OFF)
if(QW_BUILD_TESTS)
    enable_testing()
    add_executable(qwcore_tests Tests.cpp)
    target_include_directories(qwcore_tests PRIVATE ${UPSTREAM} compat)
    target_link_libraries(qwcore_tests PRIVATE qwcore)
    add_test(NAME core_roundtrips COMMAND qwcore_tests)
endif()
